open Lean Elab Command in
elab "audit_all" : command => do
  let env ← getEnv
  let mods := env.header.moduleNames
  let mut nstd := 0
  let mut total := 0
  let std : List Name := [``propext, ``Classical.choice, ``Quot.sound]
  for (n, ci) in env.constants.toList do
    let some idx := env.getModuleIdxFor? n | continue
    let m := mods[idx.toNat]!
    let ms := m.toString
    unless (ms.startsWith "PFUL" || ms.startsWith "MathlibPR" || ["Lift","Full","Gaps","Cancel","GramGeneral","ComplexGram","StinespringBoundary"].contains ms) do continue
    unless ci matches .thmInfo _ do continue
    if n.isInternal then continue
    total := total + 1
    let axs ← liftCoreM <| Lean.collectAxioms n
    let bad := axs.filter (fun a => !std.contains a)
    if bad.size > 0 then
      nstd := nstd + 1
      logInfo m!"NONSTD {ms} {n}: {bad.toList}"
  logInfo m!"theorems audited: {total}; with nonstandard axioms: {nstd}"
audit_all
